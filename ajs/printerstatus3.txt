#!/usr/bin/env bash
# printer_selfheal_symlinks.sh v1.16
# Adds: on-screen summary + JSON written to file.

set -u
VERSION="1.16"
echo "Version $VERSION------------------------------------------"
LOGGER_TAG="printer-selfheal"

BASE_DIR="/home/ajs/printer_data"
CONFIG_DIR="${BASE_DIR}/config"
NONCFG_DIR="${CONFIG_DIR}/noncfg"
LOG_DIR="${BASE_DIR}/logs"
JSON_OUT="${LOG_DIR}/selfheal.json"

NGINX_LOG_DIR="/var/log/nginx"

MOONRAKER_CONF_REAL="${NONCFG_DIR}/moonraker.conf"
MOONRAKER_CONF_LINK="${CONFIG_DIR}/moonraker.conf"

NGINX_MAINSV_SRC="/etc/nginx/sites-available/mainsail"
NGINX_MAINSV_DEST="/etc/nginx/sites-enabled/mainsail"

NGINX_MAINSAIL_ACCESS_SRC="${NGINX_LOG_DIR}/mainsail-access.log"
NGINX_MAINSAIL_ERROR_SRC="${NGINX_LOG_DIR}/mainsail-error.log"
NGINX_MAINSAIL_ACCESS_DEST="${LOG_DIR}/mainsail-access.log"
NGINX_MAINSAIL_ERROR_DEST="${LOG_DIR}/mainsail-error.log"

log_info() { logger -t "${LOGGER_TAG}" "INFO: $*"; }
log_error() { logger -t "${LOGGER_TAG}" "ERROR: $*"; }

ensure_dir_json() {
    echo "Starting Ensure Dir JSON exists..."
    local dir="$1"
    if [[ -d "$dir" ]]; then
        log_info "Directory exists: $dir"
        printf '{"type":"dir","path":"%s","status":"ok","message":"Directory exists"}\n' "$dir"
        return
    fi
    if mkdir -p "$dir"; then
        log_info "Directory created: $dir"
        printf '{"type":"dir","path":"%s","status":"created","message":"Directory created"}\n' "$dir"
    else
        log_error "Failed to create directory: $dir"
        printf '{"type":"dir","path":"%s","status":"error","message":"Failed to create directory"}\n' "$dir"
    fi
    echo "... Ending Ensure Dir JSON exists"
}

ensure_file_json() {
    echo "Starting Ensure File JSON exists..."
    local file="$1"
    local dir
    dir="$(dirname "$file")"
    mkdir -p "$dir"

    if [[ -e "$file" ]]; then
        log_info "File exists: $file"
        printf '{"type":"file","path":"%s","status":"ok","message":"File exists"}\n' "$file"
        return
    fi

    if touch "$file"; then
        log_info "File created: $file"
        printf '{"type":"file","path":"%s","status":"created","message":"File created"}\n' "$file"
    else
        log_error "Failed to create file: $file"
        printf '{"type":"file","path":"%s","status":"error","message":"Failed to create file"}\n' "$file"
    fi
    echo "... Ending Ensure File JSON exists"
}

ensure_symlink_json() {
    echo "Starting Ensure Symlink JSON exists..."
    local src="$1"
    local dest="$2"

    mkdir -p "$(dirname "$dest")"

    if [[ -L "$dest" ]]; then
        local actual
        actual="$(readlink -f "$dest" 2>/dev/null || true)"
        if [[ ! -e "$actual" ]]; then
            rm -f "$dest"
            ln -sf "$src" "$dest"
            log_info "Repaired broken symlink: $dest -> $src"
            printf '{"type":"symlink","src":"%s","dest":"%s","status":"repaired","message":"Broken symlink repaired"}\n' "$src" "$dest"
            echo "... Fixed Actual: $actual"
            echo "            dest: $dest"
			echo "             src: $src"
			return
        fi
        if [[ "$actual" != "$(readlink -f "$src")" ]]; then
            ln -sf "$src" "$dest"
            log_info "Corrected symlink: $dest -> $src"
            printf '{"type":"symlink","src":"%s","dest":"%s","status":"corrected","message":"Incorrect symlink corrected"}\n' "$src" "$dest"
            echo "... Fixed Actual: $actual"
            echo "            dest: $dest"
			echo "             src: $src"
            return
        fi
        printf '{"type":"symlink","src":"%s","dest":"%s","status":"ok","message":"Symlink verified"}\n' "$src" "$dest"
        return
    fi

    if [[ -e "$dest" ]]; then
        rm -rf "$dest"
        ln -sf "$src" "$dest"
        log_info "Replaced file with symlink: $dest -> $src"
        printf '{"type":"symlink","src":"%s","dest":"%s","status":"replaced","message":"Replaced non-symlink"}\n' "$src" "$dest"
        return
    fi

    ln -sf "$src" "$dest"
    log_info "Created symlink: $dest -> $src"
    printf '{"type":"symlink","src":"%s","dest":"%s","status":"created","message":"Symlink created"}\n' "$src" "$dest"
    echo "... Ending Ensure Symlink JSON exists"
}

extract_status() { sed -n 's/.*"status":"\([^"]*\)".*/\1/p'; }

enforce_moonraker_conf() {
    ensure_dir_json "$NONCFG_DIR"
    ensure_file_json "$MOONRAKER_CONF_REAL"
    ensure_symlink_json "$MOONRAKER_CONF_REAL" "$MOONRAKER_CONF_LINK"
}

batch_symlink_validator() {
    echo "Starting  Symlink Validator..."
    ensure_dir_json "$LOG_DIR"
    ensure_dir_json "$NGINX_LOG_DIR"

    ensure_symlink_json "$NGINX_MAINSV_SRC" "$NGINX_MAINSV_DEST"
    ensure_symlink_json "$NGINX_MAINSAIL_ACCESS_SRC" "$NGINX_MAINSAIL_ACCESS_DEST"
    ensure_symlink_json "$NGINX_MAINSAIL_ERROR_SRC" "$NGINX_MAINSAIL_ERROR_DEST"
    echo "... ending Symlink Validator..."
}

print_summary() {
    local MR_STATUS="$1"
    local SITE_STATUS="$2"
    local ACC_STATUS="$3"
    local ERR_STATUS="$4"

    echo
    echo "==================== SELF-HEAL SUMMARY ($VERSION) ===================="
    printf "%-40s %-10s\n" "Moonraker config:" "$MR_STATUS"
    printf "%-40s %-10s\n" "Nginx mainsail site:" "$SITE_STATUS"
    printf "%-40s %-10s\n" "Nginx access log link:" "$ACC_STATUS"
    printf "%-40s %-10s\n" "Nginx error log link:" "$ERR_STATUS"
    echo "==================================================================="
}


main() {
    log_info "Starting self-heal v$VERSION"

    mkdir -p "$LOG_DIR"

    # Capture JSON lines
    MR_JSON=$(enforce_moonraker_conf | tee /dev/stderr)
    SYM_JSON=$(batch_symlink_validator | tee /dev/stderr)

    
    # Extract first matching status for each category
    MR_STATUS=$(printf "%s\n" "$MR_JSON" | grep 'moonraker.conf"' | head -n1 | extract_status)
    SITE_STATUS=$(printf "%s\n" "$SYM_JSON" | grep 'sites-enabled/mainsail' | head -n1 | extract_status)
    ACC_STATUS=$(printf "%s\n" "$SYM_JSON" | grep 'mainsail-access.log' | head -n1 | extract_status)
    ERR_STATUS=$(printf "%s\n" "$SYM_JSON" | grep 'mainsail-error.log' | head -n1 | extract_status)


    # Print summary
    print_summary "$MR_STATUS" "$SITE_STATUS" "$ACC_STATUS" "$ERR_STATUS"

    # Write JSON file
    {
        echo "{"
        echo '  "moonraker_conf": ['
        echo "$MR_JSON" | sed 's/^/    /' | paste -sd',\n' -
        echo '  ],'
        echo '  "symlinks_and_logs": ['
        echo "$SYM_JSON" | sed 's/^/    /' | paste -sd',\n' -
        echo '  ]'
        echo "}"
    } > "$JSON_OUT"

    log_info "Self-heal v$VERSION complete"
}

main "$@"
